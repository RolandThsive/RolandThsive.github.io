#  paths.txt  -  where things live on this site

#  is a one-line change here and no post needs editing.
# ============================================================
#  key = full URL of the folder, no trailing slash.
#  Spaces in folder names are fine (they are encoded automatically).

# the post engine (3D and circuit viewers)   ->  Engine/viewer.js, Engine/viewer.css

#   for everything else. The health check uses it to find posts that load the engine from
#   an old address.

# .glb models                      ->  Model/<Name>.glb

# poster / thumbnail images        ->  Poster/<Name>.<PosterFormat>

# file extension of the posters: png or webp (change this when the folder is converted)
PosterFormat = webp

# per-model parameter files        ->  ModelParameters/<Name>.txt
ModelParameters = https://RolandThsive.github.io/Assets/3D_Assets/ModelParameters

# KiCad projects, one sub-folder each  ->  PCBAssets/<Project folder>/<file>
PCBAssets = https://RolandThsive.github.io/Assets/PCB_Assets

# share pictures (link previews)   ->  FeaturedImages/<postId>.jpg   (home.jpg for the home page)
#   used by the Songs page and the health check; the theme's head has its own copy of this address,
#   because Blogger builds the head on its server and cannot read this file
FeaturedImages = https://RolandThsive.github.io/Assets/PagesAndPosts_Assets/FeaturedImages

# world map, already drawn (one file per country + two whole-world files) ->  WorldMap/world-light.svg,

WorldMap = https://RolandThsive.github.io/Assets/PagesAndPosts_Assets/WorldMap

# the 404 page's animations        ->  ErrorPage/<name>.gif
#   (the 404 page carries these addresses itself, like posts carry the engine's: it has to show
#   something before anything else is read)
ErrorPage = https://RolandThsive.github.io/Assets/PagesAndPosts_Assets/404_Page

# icons                            ->  Icons/zodiac-icons/<animal>.svg|png, Icons/zodiac-REDGOLD-icons/<Animal>_REDGOLD.svg|png
#   (the Timeline page has its own copy of the red-gold zodiac drawings built in, so these are not loaded yet)
Icons = https://RolandThsive.github.io/Assets/PagesAndPosts_Assets/Icons

# the pictures used in posts, one folder per post   ->  PostImages/<Post title>/<file>
#   a safe copy: the posts themselves show the copies Blogger made when they were uploaded
PostImages = https://RolandThsive.github.io/Assets/PagesAndPosts_Assets/Post_images

# scripts posts use besides the engine   ->  Scripts/<name>.js   (the Flower Lamp's own viewer, DeskLampComp.js)
Scripts = https://RolandThsive.github.io/Assets/ThreeJS_scripts
